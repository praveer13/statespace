## Negating a rule {minutes=5}

:::instrument ins=INS-z3 cell=3 recorded=RUN-z3-starter
```python
from z3 import *

a, b, amount = Ints('a b amount')
s = Solver()
# "every allowed transfer leaves the sender at zero or above", with the guard on the wrong balance, negated
s.add(a >= 0, b >= 0, Not(ForAll([amount], Implies(And(amount > 0, amount <= b), a - amount >= 0))))
print(s.check())
```
:::

:::instrument ins=INS-z3 cell=4 recorded=RUN-z3-starter
```python
from z3 import *

a, b, amount = Ints('a b amount')
s = Solver()
# the same violation as an existential: some allowed transfer leaves the sender below zero
s.add(a >= 0, b >= 0, Exists([amount], And(amount > 0, amount <= b, a - amount < 0)))
print(s.check())
```
:::

:::instrument ins=INS-z3 cell=6 recorded=RUN-z3-starter
```python
from z3 import *

Acct, (A1, A2) = EnumSort('Acct', ['A1', 'A2'])
k = Const('k', Acct)
a1, a2, amount = Ints('a1 a2 amount')
before = lambda x: If(x == A1, a1, a2)
after = lambda x: If(x == A1, a1 - amount, a2 + amount)   # A1 sends amount to A2
s = Solver()
# every account starts at zero or above, and some account ends below zero
s.add(ForAll([k], before(k) >= 0), amount > 0, Exists([k], after(k) < 0))
print(s.check())
```
:::
