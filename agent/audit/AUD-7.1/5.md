**Statement · Z3Py**

def open_doc(exists, mayread, body):                 # Open(frank, "layoffs-memo")
    code = If(And(exists, mayread), 2, 0)            # 0 not_found, 2 page
    return code, If(code == 2, body, 0)

def observe(code, body):                             # what the proof compares

s = Solver()
s.add(rel)                                           # Rel: the worlds agree on what Acme may know
s.add(Or([a != b for a, b in zip(observe(*open_doc(*L)), observe(*open_doc(*R)))]))
